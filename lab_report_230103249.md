### harness.py
```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable

@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float

def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode("utf-8"):
        h = ((h ^ b) * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h

def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup):
        fn()
    samples = []
    for _ in range(runs):
        t0 = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - t0) / 1e6)
    samples.sort()
    med = samples[runs // 2]
    return BenchStats(samples[0], med, samples[(3 * runs) // 4] - samples[runs // 4])

def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:
    hw = mp.cpu_count()
    sig = fnv1a_hash(f"{phase}|{seed}|{hw}|{m1:.4f}|{m2:.4f}")
    return f"VERIFY_HASH[{phase}]: {sig:016X} (P_log={hw})"
```

```python
def fs_worker(shm_name: str, slot_idx: int, iters: int) -> None:
    shm = shared_memory.SharedMemory(name=shm_name)
    arr = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
    acc = arr[slot_idx]
    for i in range(iters):

        iters = M // p
```

### task2_scaling.py
```python
import math
import time
import multiprocessing as mp
from harness import run_benchmark, emit_verify_tag

SEED = 230103249
WARM, RUNS = 1, 5

def stage_a(n: int) -> float:
    """Serial prefix over first 5% of elements: 10 sin/sqrt ops each."""
    acc = 0.0
    for i in range(int(0.05 * n)):
        x = i * 1e-3 + 1.0
        for _ in range(5):                      # 5 x (sin, sqrt) = 10 ops
            x = math.sqrt(math.sin(x) ** 2 + x)
        acc += x
    return acc

def kernel(rng):
    """Stage B: 30 FMA/trig ops per element (15 x (multiply-add + sin))."""
    lo, hi = rng
    acc = 0.0
    for i in range(lo, hi):
        x = i * 1e-6
        for _ in range(15):
            x = x * 1.0000001 + 0.5
            x = math.sin(x) + 1e-3 * x
        acc += x
    return acc

def chunks(n, p):
    step = n // p
    return [(t * step, n if t == p - 1 else (t + 1) * step) for t in range(p)]

def median(xs):
    xs = sorted(xs)
    return xs[len(xs) // 2]

def measure(n, p):
    a_times = []
    pool = mp.Pool(processes=p)                 # created OUTSIDE the timed region
    def trial():
        t0 = time.perf_counter_ns()
        stage_a(n)
        a_times.append((time.perf_counter_ns() - t0) / 1e6)
        pool.map(kernel, chunks(n, p))
    try:
        s = run_benchmark(trial, WARM, RUNS)
    finally:
        pool.close(); pool.join()
    return s.median_ms, median(a_times[-RUNS:])

if __name__ == "__main__":
    plog = mp.cpu_count()
    ps = list(range(1, plog + 1))

    # ---- Strong scaling
    N = 1_200_000
    strong = {p: measure(N, p) for p in ps}
    t1, a1 = strong[1]
    print("### Table 2A: strong scaling (N = 1.2e6)")
    print("| p | T ms | S_p | E_p | marginal dS |")
    print("|---|---|---|---|---|")
    prev = 0.0
    for p in ps:
        S = t1 / strong[p][0]
        print(f"| {p} | {strong[p][0]:.1f} | {S:.2f} | {S/p:.2f} | {S-prev:+.2f} |")
        prev = S
    s_frac = a1 / t1
    s_obs = t1 / strong[plog][0]
    s_theory = 1 / (s_frac + (1 - s_frac) / plog)
    print(f"\ns = T_A(1)/T_1 = {s_frac:.4f}   S_inf = {1/s_frac:.2f}")
    print(f"Amdahl S(Plog={plog}) = {s_theory:.2f}   observed = {s_obs:.2f}")

    # ---- Weak scaling
    N0 = 200_000
    weak = {p: measure(p * N0, p) for p in ps}
    tw1 = weak[1][0]
    print("\n### Table 2B: weak scaling (N0 = 2.0e5 per worker)")
    print("| p | T ms | E_weak | alpha | S_Gustafson |")
    print("|---|---|---|---|---|")
    g_plog = None
    for p in ps:
        T, A = weak[p]
        alpha = A / T
        g = p - alpha * (p - 1)
        if p == plog: g_plog = g
        print(f"| {p} | {T:.1f} | {tw1/T:.2f} | {alpha:.4f} | {g:.2f} |")
    print()
    print(emit_verify_tag("PHASE2", SEED, s_obs, g_plog))
```

### task3_roofline_gemm.py
```python
import os
import math
import time
import multiprocessing as mp
from multiprocessing import shared_memory as shmmod
from harness import run_benchmark, emit_verify_tag

SEED = 230103249
PPHYS = 0  # set to physical core count
BUF = 64 * 10**6                 # 64 MB per buffer (matches 2 x 64 x 10^6 in formula)
FLOPS_PER_WORKER = 60_000_000
N = 256
BS_LIST = (8, 16, 32, 64, 96, 128)
REPS_WARM, REPS_RUN = 2, 7
G_WARM, G_RUNS = 1, 5            # GEMM benchmark repetitions

# ---------------- Part 3A workers ----------------
def bw_worker(t, p, barrier, out, reps, name_a, name_b, size):
    A = shmmod.SharedMemory(name=name_a)
    B = shmmod.SharedMemory(name=name_b)
    s, e = t * size // p, (t + 1) * size // p
    for r in range(reps):
        barrier.wait()
        t0 = time.perf_counter_ns()
        A.buf[s:e] = B.buf[s:e]
        out[r * p + t] = (time.perf_counter_ns() - t0) / 1e6
    A.close(); B.close()

def fma_worker(t, p, barrier, out, reps, flops):
    a, b = 0.9999999, 1e-9
    iters = flops // 32          # 16 FMAs x 2 FLOPs per iteration
    for r in range(reps):
        x0 = x1 = x2 = x3 = x4 = x5 = x6 = x7 = 1.0
        x8 = x9 = x10 = x11 = x12 = x13 = x14 = x15 = 1.0
        barrier.wait()
        t0 = time.perf_counter_ns()
        for _ in range(iters):
            x0 = x0 * a + b;   x1 = x1 * a + b;   x2 = x2 * a + b;   x3 = x3 * a + b
            x4 = x4 * a + b;   x5 = x5 * a + b;   x6 = x6 * a + b;   x7 = x7 * a + b
            x8 = x8 * a + b;   x9 = x9 * a + b;   x10 = x10 * a + b; x11 = x11 * a + b
            x12 = x12 * a + b; x13 = x13 * a + b; x14 = x14 * a + b; x15 = x15 * a + b
        out[r * p + t] = (time.perf_counter_ns() - t0) / 1e6

def run_parallel(target, p, extra):
    reps = REPS_WARM + REPS_RUN
    barrier = mp.Barrier(p)
    out = mp.Array('d', reps * p, lock=False)
    procs = [mp.Process(target=target, args=(t, p, barrier, out, reps) + extra)
             for t in range(p)]
    for pr in procs: pr.start()
    for pr in procs: pr.join()
    per_rep = sorted(max(out[r * p:(r + 1) * p]) for r in range(REPS_WARM, reps))
    return per_rep[len(per_rep) // 2]            # median ms

# ---------------- Part 3B GEMM workers ----------------
def gemm_worker(variant, t, p, names, bs):
    sa = shmmod.SharedMemory(name=names[0])
    sb = shmmod.SharedMemory(name=names[1])
    sc = shmmod.SharedMemory(name=names[2])
    A, B, C = sa.buf.cast('d'), sb.buf.cast('d'), sc.buf.cast('d')
    r0, r1 = t * N // p, (t + 1) * N // p
    for i in range(r0 * N, r1 * N):
        C[i] = 0.0
    if variant == 1:                              # naive i-j-k
        for i in range(r0, r1):
            for j in range(N):
                dot = 0.0
                for k in range(N):
                    dot += A[i * N + k] * B[k * N + j]
                C[i * N + j] = dot
    elif variant == 2:                            # i-k-j, hoisted a_ik
        for i in range(r0, r1):
            ci = i * N
            for k in range(N):
                a = A[ci + k]
                bk = k * N
                for j in range(N):
                    C[ci + j] += a * B[bk + j]
    else:                                         # blocked, micro-kernel (i,k,j)
        for ii in range(r0, r1, bs):
            i_hi = min(ii + bs, r1)
            for jj in range(0, N, bs):
                j_hi = min(jj + bs, N)
                for kk in range(0, N, bs):
                    k_hi = min(kk + bs, N)
                    for i in range(ii, i_hi):
                        ci = i * N
                        for k in range(kk, k_hi):
                            a = A[ci + k]
                            bk = k * N
                            for j in range(jj, j_hi):
                                C[ci + j] += a * B[bk + j]
    A.release(); B.release(); C.release()
    sa.close(); sb.close(); sc.close()

def run_gemm(variant, bs, names):
    procs = [mp.Process(target=gemm_worker, args=(variant, t, PPHYS, names, bs))
             for t in range(PPHYS)]
    for pr in procs: pr.start()
    for pr in procs: pr.join()

def snapshot(sc):
    m = sc.buf.cast('d')
    data = list(m[:N * N])
    m.release()
    return data

def max_diff(sc, ref):
    m = sc.buf.cast('d')
    d = max(abs(m[i] - ref[i]) for i in range(N * N))
    m.release()
    return d

if __name__ == "__main__":
    # ===== Part 3A =====
    sa_ = shmmod.SharedMemory(create=True, size=BUF)
    sb_ = shmmod.SharedMemory(create=True, size=BUF)
    try:
        sa_.buf[:BUF] = bytes(BUF)               # commit pages
        sb_.buf[:BUF] = bytes(BUF)
        t_bw = run_parallel(bw_worker, PPHYS, (sa_.name, sb_.name, BUF))
        bw = 2 * BUF / (t_bw / 1000) / 1e9
    finally:
        sa_.close(); sa_.unlink(); sb_.close(); sb_.unlink()
    t_fma = run_parallel(fma_worker, PPHYS, (FLOPS_PER_WORKER,))
    p_sus = PPHYS * FLOPS_PER_WORKER / (t_fma / 1000) / 1e9
    ridge = p_sus / bw
    print("### Part 3A: Roofline ceilings")
    print(f"BW_sustained = {bw:.3f} GB/s   (T_median = {t_bw:.2f} ms)")
    print(f"P_sustained  = {p_sus:.4f} GFLOP/s   (T_median = {t_fma:.2f} ms)")
    print(f"Ridge point I* = {ridge:.5f} FLOP/Byte")

    # ===== Part 3B =====
    size = N * N * 8
    s_a = shmmod.SharedMemory(create=True, size=size)
    s_b = shmmod.SharedMemory(create=True, size=size)
    s_c = shmmod.SharedMemory(create=True, size=size)
    names = (s_a.name, s_b.name, s_c.name)
    try:
        ma, mb = s_a.buf.cast('d'), s_b.buf.cast('d')
        for i in range(N * N):
            ma[i] = ((i * 7) % 13) / 13.0 + 0.1
            mb[i] = ((i * 5) % 11) / 11.0 - 0.2
        ma.release(); mb.release()

        rows = []
        st1 = run_benchmark(lambda: run_gemm(1, 0, names), G_WARM, G_RUNS)
        st2 = run_benchmark(lambda: run_gemm(2, 0, names), G_WARM, G_RUNS)
        ref = snapshot(s_c)                      # variant-2 result (last run)
        gf = lambda ms: 2 * N**3 / (ms / 1000) / 1e9
        rows.append(("V1 naive", "-", "-", st1, gf(st1.median_ms), "-"))
        rows.append(("V2 i-k-j", "-", "-", st2, gf(st2.median_ms), "-"))
        best_bs, best_gf, best_st = None, -1.0, None
        for bs in BS_LIST:
            st = run_benchmark(lambda: run_gemm(3, bs, names), G_WARM, G_RUNS)
            g = gf(st.median_ms)
            rows.append(("V3 tiled", bs, 3 * bs * bs * 8, st, g, f"{max_diff(s_c, ref):.1e}"))
            if g > best_gf: best_bs, best_gf, best_st = bs, g, st
    finally:
        for s in (s_a, s_b, s_c):
            s.close(); s.unlink()

    print("\n### Part 3B: GEMM (N = 256)")
    print("| variant | BS | W_tile bytes | median ms | IQR ms | GFLOP/s | max abs diff vs V2 |")
    print("|---|---|---|---|---|---|---|")
    for name, bs, w, st, g, d in rows:
        print(f"| {name} | {bs} | {w} | {st.median_ms:.1f} | {st.iqr_ms:.1f} | {g:.4f} | {d} |")
    print(f"\nBS* = {best_bs}  ({best_gf:.4f} GFLOP/s)")

    # ===== Roofline mapping =====
    def attain(I): return min(p_sus, I * bw)
    pts = [("V1 naive", 0.125, rows[0][4]),
           ("V2 i-k-j (I assumed = naive)", 0.125, rows[1][4]),
           (f"V3 BS*={best_bs}", best_bs / 12, best_gf)]
    print("\n### Roofline mapping")
    print("| variant | I (FLOP/B) | attainable GFLOP/s | achieved GFLOP/s | % of attainable |")
    print("|---|---|---|---|---|")
    for name, I, g in pts:
        a = attain(I)
        print(f"| {name} | {I:.4f} | {a:.4f} | {g:.4f} | {100*g/a:.2f}% |")
    print()
    print(emit_verify_tag("PHASE3", SEED, rows[0][4], best_gf))
```

### task4_scheduling.py
```python
import os
import math
import time
import multiprocessing as mp
from harness import run_benchmark, emit_verify_tag

SEED = 230103249
PPHYS = 0  # set to physical core count
N = 8000
WARM, RUNS = 1, 3
CHUNKS = (1, 8, 64, 512)

def work(i):
    k = 1 + int((i / 40) ** 2)
    x = 0.5 + i * 1e-4
    for _ in range(k):
        x = math.sqrt(abs(math.sin(x) * math.cos(x))) + 0.5
    return x

def static_worker(kind, t, p, tw):
    t0 = time.perf_counter_ns()
    rng = range(t * N // p, (t + 1) * N // p) if kind == "block" else range(t, N, p)
    acc = 0.0
    for i in rng:
        acc += work(i)
    tw[t] = (time.perf_counter_ns() - t0) / 1e6

def dyn_worker(t, p, tw, counter, C):
    t0 = time.perf_counter_ns()
    acc = 0.0
    while True:
        with counter.get_lock():
            s = counter.value
            counter.value = s + C
        if s >= N:
            break
        for i in range(s, min(s + C, N)):
            acc += work(i)
    tw[t] = (time.perf_counter_ns() - t0) / 1e6   # includes lock-wait time

def run_config(kind, C, p):
    tw = mp.Array('d', p, lock=False)
    counter = mp.Value('i', 0)
    if kind == "dyn":
        procs = [mp.Process(target=dyn_worker, args=(t, p, tw, counter, C)) for t in range(p)]
    else:
        procs = [mp.Process(target=static_worker, args=(kind, t, p, tw)) for t in range(p)]
    for pr in procs: pr.start()
    for pr in procs: pr.join()
    return list(tw)

def bench(kind, C):
    lams = []
    def fn():
        tw = run_config(kind, C, PPHYS)
        lams.append(max(tw) / (sum(tw) / PPHYS))
    s = run_benchmark(fn, WARM, RUNS)
    lam = sorted(lams[-RUNS:])[RUNS // 2]
    return s.median_ms, s.iqr_ms, lam

if __name__ == "__main__":
    results = []
    results.append(("Block static", "-") + bench("block", 0))
    results.append(("Cyclic static", "-") + bench("cyclic", 0))
    for C in CHUNKS:
        results.append(("Dynamic queue", C) + bench("dyn", C))

    print("| strategy | C | T_wall ms | IQR ms | lambda |")
    print("|---|---|---|---|---|")
    for name, C, t, iqr, lam in results:
        print(f"| {name} | {C} | {t:.1f} | {iqr:.1f} | {lam:.3f} |")

    dyn = [r for r in results if r[0] == "Dynamic queue"]
    best = min(dyn, key=lambda r: r[2])
    print(f"\nC* = {best[1]}  (T_wall = {best[2]:.1f} ms)")
    print(emit_verify_tag("PHASE4", SEED, results[0][2], best[2]))
```
